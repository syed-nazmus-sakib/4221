Q1 (10) — Single neuron binary classification using BCE

Q1 (10) — Single neuron binary classification using BCE

Classify a house as Expensive (1) or Not Expensive (0).

Features: size in m² (\(x_1\)), distance from city centre in km (\(x_2\)).

\[z = w_1 x_1 + w_2 x_2 + b,\qquad \hat y = \sigma(z),\qquad \sigma(z)=\frac{1}{1+e^{-z}}\]

\[L = -\bigl(y\log\hat y + (1-y)\log(1-\hat y)\bigr)\]

Hint on the paper: \(\dfrac{\partial L}{\partial z}=\hat y-y\), so

\[\frac{\partial L}{\partial w_1}=(\hat y-y)x_1,\quad \frac{\partial L}{\partial w_2}=(\hat y-y)x_2,\quad \frac{\partial L}{\partial b}=\hat y-y\]

Given:

Solution

Update inside the sample loop, not after the batch. Clip \(z\) and \(\hat y\) so exp / log cannot explode.

import math

X = [[50, 20], [60, 15], [120, 5], [150, 3]]
y = [0, 0, 1, 1]
w1, w2, b = 0.01, -0.01, 0.0
lr = 0.0001

def sigmoid(z):
    z = max(min(z, 500), -500)
    return 1 / (1 + math.exp(-z))

def forward(x1, x2):
    z = w1 * x1 + w2 * x2 + b
    return sigmoid(z)

def compute_loss(y_true, y_pred):
    eps = 1e-12
    y_pred = min(max(y_pred, eps), 1 - eps)
    return -(y_true * math.log(y_pred) + (1 - y_true) * math.log(1 - y_pred))

def compute_gradients(x1, x2, y_true, y_pred):
    dz = y_pred - y_true
    dw1 = dz * x1
    dw2 = dz * x2
    db = dz
    return dw1, dw2, db

def update_parameters(dw1, dw2, db):
    global w1, w2, b
    w1 -= lr * dw1
    w2 -= lr * dw2
    b  -= lr * db

def train(epochs):
    for epoch in range(epochs):
        total_loss = 0
        for i in range(len(X)):
            x1, x2 = X[i]
            y_true = y[i]
            y_pred = forward(x1, x2)
            total_loss += compute_loss(y_true, y_pred)
            dw1, dw2, db = compute_gradients(x1, x2, y_true, y_pred)
            update_parameters(dw1, dw2, db)
        if epoch % 20 == 0:
            print("epoch", epoch, "mean_loss =", total_loss / len(X))

train(200)
test_pred = forward(100, 8)
print("Prediction:", test_pred)

image = [
    [1, 2, 3, 0],
    [0, 1, 2, 3],
    [3, 0, 1, 2],
    [2, 3, 0, 1],
]
kernel = [
    [1, 1],
    [1, 1],
]

def convolution2d(image, kernel):
    n = len(image)
    k = len(kernel)
    out_size = n - k + 1          # S=1, P=0
    out = [[0] * out_size for _ in range(out_size)]
    for i in range(out_size):
        for j in range(out_size):
            s = 0
            for u in range(k):
                for v in range(k):
                    s += image[i + u][j + v] * kernel[u][v]
            out[i][j] = s
    return out

def relu(feature_map):
    return [[max(0, x) for x in row] for row in feature_map]

def max_pool(feature_map):
    n = len(feature_map)
    pool, stride = 2, 2
    out_size = (n - pool) // stride + 1
    out = [[0] * out_size for _ in range(out_size)]
    for i in range(out_size):
        for j in range(out_size):
            m = feature_map[i * stride][j * stride]
            for u in range(pool):
                for v in range(pool):
                    val = feature_map[i * stride + u][j * stride + v]
                    if val > m:
                        m = val
            out[i][j] = m
    return out

conv_out = convolution2d(image, kernel)
relu_out = relu(conv_out)
pool_out = max_pool(relu_out)
print(pool_out)

Q3 (20) — Simplified YOLO post-processing

Each box is [x1, y1, x2, y2, confidence]. Overlap if \(\mathrm{IoU}>0.5\).

\[\mathrm{IoU}=\frac{\text{area of overlap}}{\text{area of union}}\]

predictions = [
    [10, 10, 50, 50, 0.85],
    [100, 100, 150, 150, 0.40],
    [15, 15, 55, 55, 0.75],
]

  • (a) Remove boxes with confidence \(< 0.5\)
  • (b) Implement compute_iou(box1, box2)
  • (c) Compare the first two remaining boxes
  • (d) If IoU \(> 0.5\), keep only the higher-confidence box
  • (e) Print the final remaining boxes

Solution

predictions = [
    [10, 10, 50, 50, 0.85],
    [100, 100, 150, 150, 0.40],
    [15, 15, 55, 55, 0.75],
]

def compute_iou(box1, box2):
    x1 = max(box1[0], box2[0])
    y1 = max(box1[1], box2[1])
    x2 = min(box1[2], box2[2])
    y2 = min(box1[3], box2[3])
    inter_w = max(0, x2 - x1)
    inter_h = max(0, y2 - y1)
    inter = inter_w * inter_h
    area1 = max(0, box1[2] - box1[0]) * max(0, box1[3] - box1[1])
    area2 = max(0, box2[2] - box2[0]) * max(0, box2[3] - box2[1])
    union = area1 + area2 - inter
    if union == 0:
        return 0.0
    return inter / union

def simple_nms(predictions, threshold):
    # (a) drop low confidence
    boxes = [b for b in predictions if b[4] >= threshold]
    # (c)(d) compare remaining; keep higher if IoU > 0.5
    if len(boxes) >= 2:
        iou = compute_iou(boxes[0], boxes[1])
        if iou > 0.5:
            boxes = [boxes[0] if boxes[0][4] >= boxes[1][4] else boxes[1]]
    return boxes

result = simple_nms(predictions, 0.5)
print(result)

Full greedy NMS (safer if he adds more boxes):

def simple_nms(predictions, threshold):
    boxes = [b for b in predictions if b[4] >= threshold]
    boxes = sorted(boxes, key=lambda b: b[4], reverse=True)
    keep = []
    while boxes:
        best = boxes.pop(0)
        keep.append(best)
        boxes = [b for b in boxes if compute_iou(best, b) <= 0.5]
    return keep

import numpy as np

X = [[50, 20], [60, 15], [120, 5], [150, 3]]
y = [0, 0, 1, 1]
w1, w2, b = 0.01, -0.01, 0.0
lr = 0.0001

def sigmoid(z):
    z = np.clip(z, -500, 500)
    return 1 / (1 + np.exp(-z))

def forward(x1, x2):
    return sigmoid(w1 * x1 + w2 * x2 + b)

def compute_loss(y_true, y_pred):
    y_pred = np.clip(y_pred, 1e-12, 1 - 1e-12)
    return -(y_true * np.log(y_pred) + (1 - y_true) * np.log(1 - y_pred))

    dz = y_pred - y_true

              [-0.01,  0.01]])

import numpy as np

X = [[50., 20.], [60., 15.], [120., 5.], [150., 3.]]
Y = [[0., 0.], [0., 0.], [1., 1.], [1., 1.]]
W = np.array([[0.01, -0.01],
              [0.02, -0.03]])
b = np.zeros(2)
lr = 0.0001

def sigmoid(z):
    z = np.clip(z, -500, 500)
    return 1 / (1 + np.exp(-z))

def softmax(z):
    z = np.asarray(z, dtype=float)
    e = np.exp(z - np.max(z))
    return e / e.sum()

def bce_sum(yhat, y):
    yhat = np.clip(yhat, 1e-12, 1 - 1e-12)
    return -np.sum(y * np.log(yhat) + (1 - y) * np.log(1 - yhat))

for epoch in range(200):
    total = 0.0
    for i in range(len(X)):
        x = np.asarray(X[i])
        y = np.asarray(Y[i])
        z = W @ x + b
        yhat = sigmoid(z)
        total += bce_sum(yhat, y)
        dz = yhat - y
        W = W - lr * np.outer(dz, x)
        b = b - lr * dz
    if epoch % 20 == 0:

def prec_rec(y, yhat, t):
    pred = (yhat >= t).astype(int)
    tp = np.sum((y == 1) & (pred == 1))
    fp = np.sum((y == 0) & (pred == 1))

    n = len(image)
    m = len(image[0])
    out = [[0] * (m + 2 * p) for _ in range(n + 2 * p)]

    if padding:
        image = pad_zeros(image, padding)
    n = len(image)
    k = len(kernel)
    out_size = (n - k) // stride + 1

def max_pool_argmax(arr, pool=2, stride=2):
    n = len(arr)
    out_n = (n - pool) // stride + 1
    mx = [[0] * out_n for _ in range(out_n)]
    local = [[None] * out_n for _ in range(out_n)]
    glob = [[None] * out_n for _ in range(out_n)]
    for i in range(out_n):

            local[i][j] = (bu, bv)
            glob[i][j] = (i * stride + bu, j * stride + bv)
